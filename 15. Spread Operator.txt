*Spread Operator(...) ဖြန့်ဝေ

-Array သို့ Object ထဲက အချက်အလက်တွေကို ဖြန့်ထုတ်ပေးတဲ့ Operator ဖြစ်ပါတယ်။

-1. Spread Operator with Array
 let fruits = ["Apple", "Banana", "Orange"];
 console.log(...fruits);

-2. Copy Array
 let numbers = [1, 2, 3];
 let newNumbers = [...numbers];
 console.log(newNumbers);

-3. Combine Arrays
 let a = [1, 2, 3];
 let b = [4, 5, 6];
 let result = [...a, ...b];
 console.log(result);

-4. Spread Operator with Object
 let person = {
     name: "John",
     age: 20
 };
 let newPerson = {...person};
 console.log(newPerson);

-5. Add New Data to Object
 let person = {
    name: "John",
    age: 20
 };

 let newPerson = {
     ...person,
     city: "Yangon"
 };
 console.log(newPerson);

-6. Spread Operator in Function
 function add(a, b, c) {
    return a + b + c;
 }
 let numbers = [10, 20, 30];
 console.log(add(...numbers));
